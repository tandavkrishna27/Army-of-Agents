# E7.1 — Conversational routines Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans after explicit implementation approval, task-by-task. This file authorizes planning only.

```ts
import { z } from "zod";
import { createRoutineSchema, createRoutineTriggerSchema,
  updateRoutineSchema, updateRoutineTriggerSchema, runRoutineSchema } from "./routine.js";
const schedule = createRoutineTriggerSchema.options[0];
export const universeRoutineActionSchema = z.discriminatedUnion("action", [
  z.object({ action: z.literal("create"), routine: createRoutineSchema,
    trigger: schedule.extend({ timezone: z.string().min(1) }) }).strict(),
  z.object({ action: z.literal("update"), routineId: z.string().uuid(),
    patch: updateRoutineSchema,
    triggerUpdate: z.object({ id: z.string().uuid(), patch: updateRoutineTriggerSchema }).strict().optional(),
    baseRevisionId: z.string().uuid().nullable() }).strict(),
  z.object({ action: z.literal("pause"), routineId: z.string().uuid(),
    baseRevisionId: z.string().uuid().nullable() }).strict(),
  z.object({ action: z.literal("run"), routineId: z.string().uuid(),
    input: runRoutineSchema.omit({ source: true, idempotencyKey: true }) }).strict(),
]);
export type UniverseRoutineAction = z.infer<typeof universeRoutineActionSchema>;
export type UniverseRoutineActionResult = {
  routineId: string; revisionId: string | null; runId: string | null;
  issueId: string | null;
  state: "saved" | "paused" | "received" | "issue_created" | "coalesced" | "skipped" | "completed" | "failed";
};
```

## Migration, rollback and final evidence
