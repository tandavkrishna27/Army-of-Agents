> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans after explicit implementation approval. This is a planning addendum; no tests or migrations ran here.

```ts
import type { UniversePreferences } from "./validators/universe-preferences.js";
export function personalDeliveryRestrictions(
  p: Pick<UniversePreferences, "soundEffects" | "spokenAnnouncements">,
): { soundEffects: boolean; quieter: boolean } {
  return { soundEffects: p.soundEffects,
    quieter: p.spokenAnnouncements === "quieter" };
}
export type DeliveryLocalState = {
  dnd: boolean; soundEffects: boolean; quieter: boolean;
  activeVoiceConversation: boolean; userSpeaking: boolean; playbackEnabled: boolean;
};
// After source visibility/status checks and shared quiet-hour calculation:
export function channelPermitted(
  channel: "toast" | "sound" | "voice",
  rule: { deliveryMode: "silent" | "digest" | "realtime";
    toastEnabled: boolean; soundEnabled: boolean; voiceEnabled: boolean },
  local: DeliveryLocalState,
): boolean {
  if (local.dnd || rule.deliveryMode !== "realtime") return false;
  if (channel === "toast") return rule.toastEnabled;
  if (channel === "sound") return rule.soundEnabled && local.soundEffects;
  return rule.voiceEnabled && local.activeVoiceConversation && !local.userSpeaking
    && local.playbackEnabled && !local.quieter;
}
```
